   weekly_reports(
     id uuid primary key default gen_random_uuid()  (match how other migrations mint ids),
     group_id   -> groups(id) on delete cascade,
     student_id -> users(id) on delete cascade,
     course_id  -> courses(id) on delete cascade,
     week_start date not null            -- the Saturday, Africa/Cairo calendar date (D-66)
     status text not null check (status in ('draft','published')),
     content jsonb not null,
     generated_at timestamptz(3) not null,
     published_at timestamptz(3) null,
     published_by -> users(id) null (on delete set null),
     created_at / updated_at timestamptz(3) not null default now(),
     unique (group_id, student_id, week_start),
     check ((status = 'published') = (published_at is not null))
   ) + index (student_id, status, week_start desc) + index (group_id, week_start).

2. backend/src/reports/interfaces/weekly-report-repository.interface.ts:
   export interface WeeklyReportContent {
     attendance: { present: number; late: number; absent: number; unmarked: number; expected: number;
       sessions: { sessionId: string; title: string; scheduledAt: string;
                   status: 'present' | 'late' | 'absent' | null }[] };
     homework: { due: number; submitted: number;
       tasks: { assessmentId: string; title: string; type: string; dueAt: string; status: string;
                score: number | null; maxScore: number }[] };
   }
   export interface WeeklyReport { id; groupId; studentId; courseId; weekStart: string /* YYYY-MM-DD */;
     status: 'draft' | 'published'; content: WeeklyReportContent; generatedAt: string;
     publishedAt: string | null; publishedBy: string | null; }
   export interface WeeklyReportWeek { groupId; weekStart; drafts: number; published: number }
   export interface WeeklyReportRepository {
     /** Insert, or refresh content+generated_at of a DRAFT. Never touches a published row.
         Returns 'inserted' | 'updated' | 'skipped' (skipped = the row is published). */
     upsertDraft(input: { groupId; studentId; courseId; weekStart; content; generatedAt }): Promise<...>;
     findByGroupWeek(groupId: string, weekStart: string): Promise<WeeklyReport[]>;
     listWeeks(): Promise<WeeklyReportWeek[]>;              // newest week first, then groupId
     /** Flip every DRAFT of the group-week to published; returns the rows it flipped. */
     publishGroupWeek(groupId, weekStart, publishedBy: string, at: string): Promise<WeeklyReport[]>;
     findPublishedForStudent(studentId: string): Promise<WeeklyReport[]>;  // newest week first
   }
   export const WEEKLY_REPORT_REPOSITORY = Symbol('WEEKLY_REPORT_REPOSITORY');

3. Postgres implementation: upsertDraft is ONE statement:
   INSERT ... ON CONFLICT (group_id, student_id, week_start) DO UPDATE SET content = EXCLUDED.content,
   generated_at = EXCLUDED.generated_at, updated_at = now() WHERE weekly_reports.status = 'draft'
   RETURNING (xmax = 0) AS inserted   -- no row returned => skipped.
   (Verify the xmax trick on PG 15; if you prefer, distinguish with a CTE - but it must stay one
   round trip and the published guard must be in SQL.) week_start returned as 'YYYY-MM-DD' text (no
   timezone shift - select it as week_start::text). publishGroupWeek is one UPDATE ... WHERE
   status='draft' RETURNING *. Joins the ambient transaction like every other repository.

4. In-memory implementation with identical semantics; every read returns COPIES (CLAUDE.md §9: a
   before snapshot must not alias its after).
